cmake_minimum_required(VERSION 3.16)
if(COMMAND idf_component_register)
    if(NOT IDF_TARGET STREQUAL "esp32s3")
        message(FATAL_ERROR "libespnow ABI is pinned to esp32s3")
    endif()
    idf_component_register(SRCS src/api.c src/core.c src/rate.c src/offchannel_abi.c src/abi_idf.c
                           INCLUDE_DIRS include REQUIRES esp_wifi)
    target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
    set_target_properties(${COMPONENT_LIB} PROPERTIES OUTPUT_NAME espnow)
else()
    project(libespnow VERSION 0.1.1 LANGUAGES C)
    enable_testing()
    option(LIBESPNOW_SANITIZERS "Enable host AddressSanitizer/UBSan" ON)
    add_library(espnow STATIC src/api.c src/core.c src/rate.c src/offchannel_abi.c)
    target_include_directories(espnow PUBLIC include)
    target_include_directories(espnow SYSTEM PUBLIC spec/public-sdk tests/host)
    target_compile_features(espnow PUBLIC c_std_11)
    target_compile_options(espnow PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
    add_executable(espnow_tests tests/test_espnow.c tests/model_test.c tests/mock_platform.c)
    target_link_libraries(espnow_tests PRIVATE espnow)
    target_compile_options(espnow_tests PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes)
    if(LIBESPNOW_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        foreach(target espnow espnow_tests)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endforeach()
    endif()
    add_test(NAME espnow_semantics COMMAND espnow_tests)
endif()
